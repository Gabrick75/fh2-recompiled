# Prototype: FH2 on the ShiftGlue SDK (evaluation only, not part of FH2 Recompiled).
cmake_minimum_required(VERSION 3.25)
project(fh2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(rexglue_proto.cmake)
set(FH2_SOURCES src/main.cpp src/fh2_teste_entrada.cpp src/fh2_captura.cpp src/fh2_correcoes.cpp src/fh2_toque.cpp src/fh2_medidor.cpp)
if(ANDROID)
    add_library(fh2 SHARED ${FH2_SOURCES})
    set_target_properties(fh2 PROPERTIES OUTPUT_NAME main)
    target_link_options(fh2 PRIVATE -Wl,-Bsymbolic-functions)
else()
    add_executable(fh2 ${FH2_SOURCES})
endif()
rexglue_setup_target(fh2 GPU_PLUGINS fh1)
if(TARGET imgui)
    target_include_directories(fh2 PRIVATE $<TARGET_PROPERTY:imgui,INTERFACE_INCLUDE_DIRECTORIES>)
endif()
if(TARGET renderdoc)
    target_include_directories(fh2 PRIVATE $<TARGET_PROPERTY:renderdoc,INTERFACE_INCLUDE_DIRECTORIES>)
endif()
